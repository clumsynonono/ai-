# 第3章：第一个智能合约

在本章中，我们将深入学习智能合约的结构，编写并部署完整的合约示例。

## 3.1 智能合约基本结构

一个标准的 Solidity 合约包含以下元素：

```solidity
// SPDX-License-Identifier: MIT          // 1. 许可证标识
pragma solidity ^0.8.0;                  // 2. 编译器版本

// 3. 导入其他合约（可选）
// import "./OtherContract.sol";

// 4. 合约声明
contract SimpleStorage {

    // 5. 状态变量
    uint256 private storedData;
    address public owner;

    // 6. 事件
    event DataStored(uint256 newValue);

    // 7. 修饰器
    modifier onlyOwner() {
        require(msg.sender == owner, "Not the owner");
        _;
    }

    // 8. 构造函数
    constructor() {
        owner = msg.sender;
    }

    // 9. 函数
    function set(uint256 _value) public onlyOwner {
        storedData = _value;
        emit DataStored(_value);
    }

    function get() public view returns (uint256) {
        return storedData;
    }
}
```

### 详细解析

#### 1. SPDX 许可证标识
```solidity
// SPDX-License-Identifier: MIT
```
- 声明代码的开源许可证
- 常用许可证：MIT, GPL-3.0, Apache-2.0
- 不写会有编译警告

#### 2. Pragma 指令
```solidity
pragma solidity ^0.8.0;   // >= 0.8.0 且 < 0.9.0
pragma solidity 0.8.19;   // 精确版本
pragma solidity >=0.8.0 <0.9.0;  // 范围版本
```

#### 3. 导入语句
```solidity
import "./MyContract.sol";                    // 相对路径
import "@openzeppelin/contracts/token/ERC20/ERC20.sol";  // npm 包
```

## 3.2 状态变量详解

状态变量存储在区块链上，永久保存。

### 可见性修饰符
```solidity
contract VisibilityExample {
    uint256 public publicVar = 1;      // 任何人可读，自动生成 getter
    uint256 private privateVar = 2;    // 仅合约内部可访问
    uint256 internal internalVar = 3;  // 合约内部和继承合约可访问
    // 注意：没有 external 状态变量
}
```

### 数据位置
```solidity
contract DataLocation {
    uint256[] public numbers;  // storage（默认）

    function processArray(uint256[] memory _nums) public {
        // memory：临时存储，函数执行后清除
        uint256[] memory tempArray = new uint256[](5);
    }

    function getString() public pure returns (string memory) {
        // string 必须指定 memory 或 storage
        return "Hello";
    }
}
```

## 3.3 函数详解

### 函数结构
```solidity
function functionName(参数类型 参数名)
    可见性
    状态可变性
    修饰器
    returns (返回类型)
{
    // 函数体
}
```

### 可见性修饰符
```solidity
contract FunctionVisibility {
    // public：任何地方都可调用
    function publicFunc() public pure returns (string memory) {
        return "Public";
    }

    // external：仅外部调用（不能被合约内部调用）
    function externalFunc() external pure returns (string memory) {
        return "External";
    }

    // internal：仅合约内部和继承合约可调用
    function internalFunc() internal pure returns (string memory) {
        return "Internal";
    }

    // private：仅合约内部可调用
    function privateFunc() private pure returns (string memory) {
        return "Private";
    }

    // 调用示例
    function callFunctions() public view returns (string memory) {
        // publicFunc();     // ✅ 可以
        // this.externalFunc();  // ✅ 通过 this 调用
        // externalFunc();   // ❌ 不可以直接调用
        // internalFunc();   // ✅ 可以
        // privateFunc();    // ✅ 可以
        return "OK";
    }
}
```

### 状态可变性
```solidity
contract StateMutability {
    uint256 public value = 10;

    // view：只读，不修改状态
    function getValue() public view returns (uint256) {
        return value;  // 可以读取
    }

    // pure：不读取也不修改状态
    function add(uint256 a, uint256 b) public pure returns (uint256) {
        return a + b;  // 纯计算
    }

    // payable：可以接收 ETH
    function deposit() public payable {
        // 可以接收 msg.value
    }

    // 默认（无修饰符）：可以修改状态
    function setValue(uint256 _value) public {
        value = _value;  // 修改状态变量
    }
}
```

## 3.5 修饰器 (Modifier)

修饰器用于在函数执行前后添加检查或逻辑。

```solidity
contract ModifierExample {
    address public owner;
    bool public paused = false;

    constructor() {
        owner = msg.sender;
    }

    // 定义修饰器
    modifier onlyOwner() {
        require(msg.sender == owner, "Not the owner");
        _;  // 占位符，代表被修饰函数的代码
    }

    modifier whenNotPaused() {
        require(!paused, "Contract is paused");
        _;
    }

    modifier validAddress(address _addr) {
        require(_addr != address(0), "Invalid address");
        _;
    }

    // 使用修饰器
    function pause() public onlyOwner {
        paused = true;
    }

    function unpause() public onlyOwner {
        paused = false;
    }

    // 多个修饰器按顺序执行
    function criticalFunction(address _addr)
        public
        onlyOwner
        whenNotPaused
        validAddress(_addr)
    {
        // 函数逻辑
    }
}
```

## 3.6 事件 (Events)

事件用于记录链上活动，供前端监听。

```solidity
contract EventExample {
    // 定义事件
    event Transfer(address indexed from, address indexed to, uint256 amount);
    event Deposit(address indexed user, uint256 amount, uint256 timestamp);

    mapping(address => uint256) public balances;

    function transfer(address _to, uint256 _amount) public {
        require(balances[msg.sender] >= _amount, "Insufficient balance");

        balances[msg.sender] -= _amount;
        balances[_to] += _amount;

        // 触发事件
        emit Transfer(msg.sender, _to, _amount);
    }

    function deposit() public payable {
        balances[msg.sender] += msg.value;

        // 最多 3 个 indexed 参数
        emit Deposit(msg.sender, msg.value, block.timestamp);
    }
}
```

### indexed 参数
- 最多 3 个 indexed 参数
- indexed 参数可以用于过滤和搜索
- 非 indexed 参数存储在事件的 data 部分

## 3.7 完整示例：计数器合约

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

/**
 * @title Counter
 * @dev 一个简单的计数器合约，演示基本功能
 */
contract Counter {
    // 状态变量
    uint256 private count;
    address public owner;

    // 事件
    event CountChanged(uint256 newCount, address changedBy);
    event OwnershipTransferred(address indexed previousOwner, address indexed newOwner);

    // 修饰器
    modifier onlyOwner() {
        require(msg.sender == owner, "Only owner can call this");
        _;
    }

    // 构造函数
    constructor() {
        owner = msg.sender;
        count = 0;
    }

    // 增加计数
    function increment() public {
        count += 1;
        emit CountChanged(count, msg.sender);
    }

    // 减少计数
    function decrement() public {
        require(count > 0, "Counter cannot be negative");
        count -= 1;
        emit CountChanged(count, msg.sender);
    }

    // 获取当前计数（view 函数）
    function getCount() public view returns (uint256) {
        return count;
    }

    // 重置计数（仅 owner）
    function reset() public onlyOwner {
        count = 0;
        emit CountChanged(0, msg.sender);
    }

    // 转移所有权
    function transferOwnership(address newOwner) public onlyOwner {
        require(newOwner != address(0), "New owner cannot be zero address");
        address oldOwner = owner;
        owner = newOwner;
        emit OwnershipTransferred(oldOwner, newOwner);
    }
}
```

## 3.9 常用全局变量

```solidity
contract GlobalVariables {
    function getGlobalVars() public view returns (
        address,  // msg.sender
        uint256,  // msg.value
        uint256,  // block.timestamp
        uint256   // block.number
    ) {
        return (
            msg.sender,        // 调用者地址
            msg.value,         // 发送的 ETH 数量（wei）
            block.timestamp,   // 当前区块时间戳
            block.number       // 当前区块号
        );
    }

    function getMoreVars() public view returns (address, uint256, uint256) {
        return (
            tx.origin,         // 交易发起者（不推荐用于权限控制）
            block.difficulty,  // 区块难度（在 PoS 后为 0）
            block.gaslimit     // 区块 gas 限制
        );
    }
}
```
