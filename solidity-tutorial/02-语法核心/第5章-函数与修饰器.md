# 第5章：函数与修饰器

函数是智能合约的核心，本章将深入学习函数的各种特性和最佳实践。

## 5.1 函数声明

### 完整的函数语法
```solidity
function functionName(参数类型 参数名)
    <visibility>        // public, private, internal, external
    <stateMutability>   // pure, view, payable
    <modifiers>         // 自定义修饰器
    returns (返回类型)
{
    // 函数体
}
```

### 示例
```solidity
contract FunctionBasics {
    uint256 public value;

    function setValue(uint256 _value)
        public                    // 可见性
        returns (uint256)         // 返回类型
    {
        value = _value;
        return value;
    }
}
```

## 5.2 可见性修饰符详解

### 四种可见性
```solidity
contract VisibilityDemo {
    uint256 private privateVar = 1;

    // 1. public：所有地方都可调用
    function publicFunc() public pure returns (string memory) {
        return "Public: 外部和内部都可调用";
    }

    // 2. external：只能从外部调用
    function externalFunc() external pure returns (string memory) {
        return "External: 仅外部调用";
    }

    // 3. internal：仅内部和子合约可调用
    function internalFunc() internal pure returns (string memory) {
        return "Internal: 内部和继承合约";
    }

    // 4. private：仅当前合约内部可调用
    function privateFunc() private pure returns (string memory) {
        return "Private: 仅当前合约";
    }

    // 调用演示
    function testVisibility() public view returns (string memory) {
        // publicFunc();           // ✅ 可以
        // this.publicFunc();      // ✅ 也可以（external 调用）

        // externalFunc();         // ❌ 不可以
        // this.externalFunc();    // ✅ 可以（通过 external 接口）

        // internalFunc();         // ✅ 可以
        // privateFunc();          // ✅ 可以

        return "Test complete";
    }
}

// 继承合约
contract Child is VisibilityDemo {
    function testInheritance() public pure returns (string memory) {
        // publicFunc();      // ✅ 可以
        // internalFunc();    // ✅ 可以
        // privateFunc();     // ❌ 不可以
        return "Inheritance test";
    }
}
```

### 可见性选择指南
```
┌─────────────────────────────────────────────────────┐
│ 使用场景                │ 推荐可见性              │
├─────────────────────────────────────────────────────┤
│ 用户直接调用            │ external（省 Gas）      │
│ 内部和外部都要调用      │ public                   │
│ 仅内部和子合约使用      │ internal                 │
│ 仅当前合约使用          │ private                  │
└─────────────────────────────────────────────────────┘
```

## 5.3 状态可变性

### pure - 纯函数
不读取也不修改状态
```solidity
contract PureFunctions {
    // ✅ 纯计算
    function add(uint256 a, uint256 b) public pure returns (uint256) {
        return a + b;
    }

    // ✅ 字符串操作
    function concatenate(string memory a, string memory b)
        public
        pure
        returns (string memory)
    {
        return string(abi.encodePacked(a, b));
    }

    // ✅ 哈希计算
    function hash(string memory text) public pure returns (bytes32) {
        return keccak256(abi.encodePacked(text));
    }
}
```

### view - 视图函数
只读取状态，不修改
```solidity
contract ViewFunctions {
    uint256 public balance = 100;
    address public owner;

    constructor() {
        owner = msg.sender;
    }

    // ✅ 读取状态变量
    function getBalance() public view returns (uint256) {
        return balance;
    }

    // ✅ 读取 msg.sender
    function isOwner() public view returns (bool) {
        return msg.sender == owner;
    }

    // ✅ 读取 block 信息
    function getBlockInfo() public view returns (uint256, uint256) {
        return (block.timestamp, block.number);
    }

    // ✅ 调用其他 view 函数
    function complexView() public view returns (uint256) {
        uint256 bal = getBalance();
        return bal * 2;
    }
}
```

### payable - 可支付函数
可以接收 ETH
```solidity
contract PayableFunctions {
    mapping(address => uint256) public balances;

    // 接收 ETH
    function deposit() public payable {
        require(msg.value > 0, "Must send ETH");
        balances[msg.sender] += msg.value;
    }

    // 带参数的 payable 函数
    function buyTokens(uint256 amount) public payable {
        require(msg.value >= amount * 1 ether, "Insufficient ETH");
        // 购买逻辑
    }

    // 查看合约余额
    function getContractBalance() public view returns (uint256) {
        return address(this).balance;
    }

    // 提取 ETH
    function withdraw(uint256 amount) public {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        balances[msg.sender] -= amount;

        // 转账方式（推荐）
        (bool success, ) = msg.sender.call{value: amount}("");
        require(success, "Transfer failed");
    }
}
```

### 默认（可修改状态）
```solidity
contract StateMutating {
    uint256 public counter;

    // 修改状态变量
    function increment() public {
        counter += 1;
    }

    // 可以做任何操作
    function reset() public {
        counter = 0;
        emit Reset(msg.sender);
    }

    event Reset(address by);
}
```

## 5.4 返回值

### 单个返回值
```solidity
contract SingleReturn {
    function getNumber() public pure returns (uint256) {
        return 42;
    }

    function getString() public pure returns (string memory) {
        return "Hello";
    }
}
```

### 多个返回值
```solidity
contract MultipleReturns {
    // 方式 1：命名返回值
    function getValues1() public pure returns (uint256 num, string memory text, bool flag) {
        num = 42;
        text = "Hello";
        flag = true;
        // 自动返回，也可以显式 return
    }

    // 方式 2：return 语句
    function getValues2() public pure returns (uint256, string memory, bool) {
        return (42, "Hello", true);
    }

    // 方式 3：混合
    function getValues3() public pure returns (uint256 num, string memory, bool) {
        num = 42;
        return (num, "Hello", true);
    }

    // 接收返回值
    function useReturns() public pure {
        // 接收所有返回值
        (uint256 num, string memory text, bool flag) = getValues2();

        // 只接收部分返回值
        (uint256 num2, , bool flag2) = getValues2();

        // 只接收第一个
        (uint256 num3, , ) = getValues2();
    }
}
```

### 返回结构体和数组
```solidity
contract ComplexReturns {
    struct Person {
        string name;
        uint256 age;
    }

    // 返回结构体
    function getPerson() public pure returns (Person memory) {
        return Person("Alice", 25);
    }

    // 返回数组
    function getArray() public pure returns (uint256[] memory) {
        uint256[] memory arr = new uint256[](3);
        arr[0] = 1;
        arr[1] = 2;
        arr[2] = 3;
        return arr;
    }

    // 返回多个数组
    function getMultipleArrays()
        public
        pure
        returns (uint256[] memory, string[] memory)
    {
        uint256[] memory numbers = new uint256[](2);
        numbers[0] = 1;
        numbers[1] = 2;

        string[] memory names = new string[](2);
        names[0] = "Alice";
        names[1] = "Bob";

        return (numbers, names);
    }
}
```

## 5.5 函数修饰器 (Modifier)

### 基础修饰器
```solidity
contract ModifierBasics {
    address public owner;
    bool public paused;

    constructor() {
        owner = msg.sender;
    }

    // 定义修饰器
    modifier onlyOwner() {
        require(msg.sender == owner, "Not owner");
        _;  // 占位符：执行被修饰的函数
    }

    modifier whenNotPaused() {
        require(!paused, "Contract paused");
        _;
    }

    // 使用修饰器
    function pause() public onlyOwner {
        paused = true;
    }

    function unpause() public onlyOwner {
        paused = false;
    }

    function sensitiveOperation() public onlyOwner whenNotPaused {
        // 只有 owner 且未暂停时才能执行
    }
}
```

### 带参数的修饰器
```solidity
contract ParameterizedModifiers {
    mapping(address => uint256) public balances;

    modifier minAmount(uint256 _min) {
        require(msg.value >= _min, "Insufficient amount");
        _;
    }

    modifier validAddress(address _addr) {
        require(_addr != address(0), "Invalid address");
        require(_addr != address(this), "Cannot be contract");
        _;
    }

    modifier costs(uint256 _cost) {
        require(msg.value >= _cost, "Insufficient payment");
        _;
        // 退还多余的 ETH
        if (msg.value > _cost) {
            payable(msg.sender).transfer(msg.value - _cost);
        }
    }

    // 使用
    function deposit() public payable minAmount(0.1 ether) {
        balances[msg.sender] += msg.value;
    }

    function transfer(address _to, uint256 _amount)
        public
        validAddress(_to)
    {
        require(balances[msg.sender] >= _amount, "Insufficient balance");
        balances[msg.sender] -= _amount;
        balances[_to] += _amount;
    }
}
```

### 高级修饰器模式
```solidity
contract AdvancedModifiers {
    uint256 private lockCounter;

    // 防重入锁
    modifier nonReentrant() {
        lockCounter += 1;
        uint256 localCounter = lockCounter;
        _;
        require(localCounter == lockCounter, "Reentrant call");
    }

    // 前后都执行代码
    modifier logged() {
        emit FunctionCalled(msg.sender, "Start");
        _;  // 执行函数
        emit FunctionCalled(msg.sender, "End");
    }

    event FunctionCalled(address caller, string stage);

    // 多个修饰器的执行顺序
    modifier first() {
        // 1. first 的前置代码
        _;  // 3. 执行 second
        // 6. first 的后置代码
    }

    modifier second() {
        // 2. second 的前置代码
        _;  // 4. 执行函数体
        // 5. second 的后置代码
    }

    function testOrder() public first second {
        // 函数体
    }
}
```

## 5.6 特殊函数

### 构造函数
```solidity
contract ConstructorExamples {
    address public owner;
    uint256 public immutable creationTime;
    string public name;

    // 基本构造函数
    constructor(string memory _name) {
        owner = msg.sender;
        creationTime = block.timestamp;
        name = _name;
    }

    // 带 payable 的构造函数
    constructor(string memory _name) payable {
        require(msg.value >= 1 ether, "Must send 1 ETH");
        owner = msg.sender;
        name = _name;
    }
}
```

### receive 函数
接收纯 ETH 转账（无数据）
```solidity
contract ReceiveExample {
    event Received(address sender, uint256 amount);

    // 接收 ETH
    receive() external payable {
        emit Received(msg.sender, msg.value);
    }

    function getBalance() public view returns (uint256) {
        return address(this).balance;
    }
}
```

### fallback 函数
接收带数据的调用或不存在的函数调用
```solidity
contract FallbackExample {
    event FallbackCalled(address sender, uint256 value, bytes data);

    // fallback：处理未知函数调用
    fallback() external payable {
        emit FallbackCalled(msg.sender, msg.value, msg.data);
    }

    // 如果只想处理调用，不接收 ETH，去掉 payable
    fallback() external {
        // 处理调用但不接收 ETH
    }
}
```

### receive vs fallback
```solidity
contract ReceiveVsFallback {
    event Log(string func, address sender, uint256 value, bytes data);

    // receive：纯 ETH 转账时调用
    receive() external payable {
        emit Log("receive", msg.sender, msg.value, "");
    }

    // fallback：其他情况
    fallback() external payable {
        emit Log("fallback", msg.sender, msg.value, msg.data);
    }

    /*
    调用流程：
    发送 ETH？
        │
        ├─ 是 → msg.data 为空？
        │         │
        │         ├─ 是 → receive() 存在？
        │         │        ├─ 是 → 调用 receive()
        │         │        └─ 否 → 调用 fallback()
        │         │
        │         └─ 否 → 调用 fallback()
        │
        └─ 否 → 调用 fallback()
    */
}
```

## 5.7 函数重载

```solidity
contract FunctionOverloading {
    // 重载：相同函数名，不同参数
    function getValue() public pure returns (uint256) {
        return 1;
    }

    function getValue(uint256 _num) public pure returns (uint256) {
        return _num;
    }

    function getValue(uint256 _num, uint256 _multiplier)
        public
        pure
        returns (uint256)
    {
        return _num * _multiplier;
    }

    // 调用重载函数
    function testOverload() public pure {
        uint256 a = getValue();           // 调用第一个
        uint256 b = getValue(5);          // 调用第二个
        uint256 c = getValue(5, 2);       // 调用第三个
    }

    // ⚠️ 注意：参数类型必须不同，仅返回值不同不算重载
    // function getValue() public pure returns (string memory) { }  // ❌ 错误
}
```

## 📝 本章小结

- ✅ 可见性：public, external, internal, private
- ✅ 状态可变性：pure, view, payable
- ✅ 修饰器：权限控制和代码复用
- ✅ 特殊函数：constructor, receive, fallback
- ✅ 函数重载和返回值

## 🎯 最佳实践

1. **external vs public**：仅外部调用时用 external（省 Gas）
2. **view/pure**：不修改状态时必须声明，帮助编译器优化
3. **修饰器**：用于权限检查和防重入
4. **receive/fallback**：正确处理 ETH 接收
5. **命名返回值**：提高代码可读性
