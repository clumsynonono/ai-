# 第6章：控制流程与错误处理

本章学习 Solidity 中的条件语句、循环以及错误处理机制。

## 6.1 条件语句

### if-else 语句
```solidity
contract ConditionalStatements {
    function checkNumber(uint256 _num) public pure returns (string memory) {
        if (_num > 100) {
            return "Greater than 100";
        } else if (_num > 50) {
            return "Greater than 50";
        } else {
            return "50 or less";
        }
    }

    // 单行 if
    function simpleCheck(uint256 _num) public pure returns (bool) {
        if (_num > 10) return true;
        return false;
    }

    // 嵌套 if
    function nestedCondition(uint256 _num) public pure returns (string memory) {
        if (_num > 0) {
            if (_num % 2 == 0) {
                return "Positive and even";
            } else {
                return "Positive and odd";
            }
        } else {
            return "Zero or negative";
        }
    }
}
```

### 三元运算符
```solidity
contract TernaryOperator {
    function max(uint256 a, uint256 b) public pure returns (uint256) {
        return a > b ? a : b;
    }

    function isEven(uint256 _num) public pure returns (string memory) {
        return _num % 2 == 0 ? "Even" : "Odd";
    }

    // 嵌套三元运算符（不推荐，可读性差）
    function grade(uint256 score) public pure returns (string memory) {
        return score >= 90 ? "A" :
               score >= 80 ? "B" :
               score >= 70 ? "C" : "F";
    }
}
```

## 6.2 循环语句

### for 循环
```solidity
contract ForLoops {
    uint256[] public numbers;

    // 基础 for 循环
    function addNumbers(uint256 _count) public {
        for (uint256 i = 0; i < _count; i++) {
            numbers.push(i);
        }
    }

    // 计算数组总和
    function sum() public view returns (uint256) {
        uint256 total = 0;
        for (uint256 i = 0; i < numbers.length; i++) {
            total += numbers[i];
        }
        return total;
    }

    // 倒序循环
    function reverseLoop() public view returns (uint256) {
        uint256 count = 0;
        for (uint256 i = numbers.length; i > 0; i--) {
            count++;
        }
        return count;
    }

    // 步长为 2
    function skipLoop() public pure returns (uint256) {
        uint256 sum = 0;
        for (uint256 i = 0; i < 10; i += 2) {
            sum += i;  // 0 + 2 + 4 + 6 + 8 = 20
        }
        return sum;
    }
}
```

### while 循环
```solidity
contract WhileLoops {
    // 基础 while 循环
    function whileExample() public pure returns (uint256) {
        uint256 i = 0;
        uint256 sum = 0;

        while (i < 10) {
            sum += i;
            i++;
        }
        return sum;  // 0+1+...+9 = 45
    }

    // 查找第一个满足条件的数
    function findFirst(uint256[] memory arr, uint256 target)
        public
        pure
        returns (int256)
    {
        uint256 i = 0;
        while (i < arr.length) {
            if (arr[i] == target) {
                return int256(i);  // 找到了，返回索引
            }
            i++;
        }
        return -1;  // 未找到
    }
}
```

### do-while 循环
```solidity
contract DoWhileLoops {
    // do-while：至少执行一次
    function doWhileExample() public pure returns (uint256) {
        uint256 i = 0;
        uint256 sum = 0;

        do {
            sum += i;
            i++;
        } while (i < 10);

        return sum;  // 45
    }

    // 即使条件不满足，也会执行一次
    function alwaysOnce() public pure returns (uint256) {
        uint256 result = 0;

        do {
            result = 100;
        } while (false);  // 条件为假，但已执行一次

        return result;  // 100
    }
}
```

### continue 和 break
```solidity
contract BreakContinue {
    // continue：跳过当前迭代
    function sumEvenNumbers() public pure returns (uint256) {
        uint256 sum = 0;

        for (uint256 i = 0; i < 10; i++) {
            if (i % 2 != 0) {
                continue;  // 跳过奇数
            }
            sum += i;  // 只累加偶数
        }
        return sum;  // 0 + 2 + 4 + 6 + 8 = 20
    }

    // break：终止循环
    function findInArray(uint256[] memory arr, uint256 target)
        public
        pure
        returns (bool)
    {
        for (uint256 i = 0; i < arr.length; i++) {
            if (arr[i] == target) {
                return true;  // 找到后立即返回
            }
        }
        return false;
    }

    // 组合使用
    function complexLoop() public pure returns (uint256) {
        uint256 count = 0;

        for (uint256 i = 0; i < 100; i++) {
            if (i < 10) continue;      // 跳过前 10 个
            if (i > 50) break;         // 超过 50 停止
            if (i % 2 == 0) count++;   // 计数偶数
        }
        return count;  // 10-50 之间的偶数个数
    }
}
```

## 6.3 循环的 Gas 优化

⚠️ **循环是 Gas 消耗大户，必须谨慎使用！**

### 避免无限循环
```solidity
contract GasOptimization {
    uint256[] public data;

    // ❌ 危险：可能耗尽 Gas
    function badLoop() public view {
        for (uint256 i = 0; i < data.length; i++) {
            // 如果 data 很大，会超出 Gas 限制
        }
    }

    // ✅ 好：限制循环次数
    function goodLoop(uint256 maxIterations) public view {
        uint256 length = data.length < maxIterations
            ? data.length
            : maxIterations;

        for (uint256 i = 0; i < length; i++) {
            // 最多执行 maxIterations 次
        }
    }

    // ✅ 更好：缓存 length（节省 Gas）
    function betterLoop() public view {
        uint256 length = data.length;  // 缓存 length
        for (uint256 i = 0; i < length; i++) {
            // 每次不用读取 storage
        }
    }

    // ✅ 最佳：使用 unchecked（0.8.0+）
    function bestLoop() public view returns (uint256) {
        uint256 length = data.length;
        uint256 sum = 0;

        for (uint256 i = 0; i < length;) {
            sum += data[i];
            unchecked { i++; }  // 节省溢出检查的 Gas
        }
        return sum;
    }
}
```

### 循环最佳实践
```solidity
contract LoopBestPractices {
    // ✅ 1. 避免在循环中修改状态变量
    uint256[] public items;
    uint256 public totalProcessed;

    // 差的做法
    function badProcess() public {
        for (uint256 i = 0; i < items.length; i++) {
            totalProcessed++;  // 每次写 storage（贵！）
        }
    }

    // 好的做法
    function goodProcess() public {
        uint256 temp = totalProcessed;  // 读一次
        for (uint256 i = 0; i < items.length; i++) {
            temp++;  // 修改 memory（便宜）
        }
        totalProcessed = temp;  // 写一次
    }

    // ✅ 2. 使用映射而不是数组遍历
    mapping(address => bool) public whitelist;
    address[] public whitelistArray;  // 避免使用

    // 差：需要遍历数组
    function isWhitelisted1(address _user) public view returns (bool) {
        for (uint256 i = 0; i < whitelistArray.length; i++) {
            if (whitelistArray[i] == _user) return true;
        }
        return false;
    }

    // 好：O(1) 查询
    function isWhitelisted2(address _user) public view returns (bool) {
        return whitelist[_user];
    }
}
```

## 6.4 错误处理

Solidity 提供三种错误处理方式：`require`、`assert`、`revert`。

### require
用于验证输入和条件
```solidity
contract RequireExample {
    mapping(address => uint256) public balances;

    function withdraw(uint256 _amount) public {
        // 验证条件，失败则回滚并返回消息
        require(_amount > 0, "Amount must be greater than zero");
        require(balances[msg.sender] >= _amount, "Insufficient balance");

        balances[msg.sender] -= _amount;
        payable(msg.sender).transfer(_amount);
    }

    // 多个 require
    function transfer(address _to, uint256 _amount) public {
        require(_to != address(0), "Invalid recipient");
        require(_to != msg.sender, "Cannot transfer to self");
        require(_amount > 0, "Amount must be positive");
        require(balances[msg.sender] >= _amount, "Insufficient balance");

        balances[msg.sender] -= _amount;
        balances[_to] += _amount;
    }

    // require 还会退还剩余 Gas
}
```

### assert
用于检查内部错误和不变量
```solidity
contract AssertExample {
    uint256 public totalSupply = 1000;
    mapping(address => uint256) public balances;

    function transfer(address _to, uint256 _amount) public {
        require(balances[msg.sender] >= _amount, "Insufficient balance");

        uint256 previousSum = balances[msg.sender] + balances[_to];

        balances[msg.sender] -= _amount;
        balances[_to] += _amount;

        // 断言：总和应该不变（检查内部错误）
        assert(balances[msg.sender] + balances[_to] == previousSum);
    }

    // assert 失败会消耗所有剩余 Gas（0.8.0 后改为 revert）
    function invariantCheck() public view {
        // 检查不变量
        assert(totalSupply >= 0);  // 这个永远为真（示例）
    }
}
```

### revert
直接回滚交易
```solidity
contract RevertExample {
    enum Status { Pending, Approved, Rejected }

    function processStatus(Status _status) public pure {
        if (_status == Status.Pending) {
            revert("Cannot process pending status");
        } else if (_status == Status.Rejected) {
            revert("Status already rejected");
        }

        // 处理 Approved 状态
    }

    // 带数据的 revert
    function advancedRevert(uint256 _value) public pure {
        if (_value < 100) {
            revert("Value too low");
        }
    }

    // 无消息的 revert
    function simpleRevert() public pure {
        revert();  // 仅回滚，无错误消息
    }
}
```

### 自定义错误（0.8.4+，推荐）
```solidity
contract CustomErrors {
    // 定义自定义错误
    error InsufficientBalance(uint256 available, uint256 required);
    error InvalidAddress(address addr);
    error Unauthorized(address caller);

    mapping(address => uint256) public balances;
    address public owner;

    constructor() {
        owner = msg.sender;
    }

    // 使用自定义错误
    function withdraw(uint256 _amount) public {
        if (msg.sender != owner) {
            revert Unauthorized(msg.sender);
        }

        uint256 balance = balances[msg.sender];
        if (balance < _amount) {
            revert InsufficientBalance(balance, _amount);
        }

        balances[msg.sender] -= _amount;
    }

    function transfer(address _to, uint256 _amount) public {
        if (_to == address(0)) {
            revert InvalidAddress(_to);
        }

        if (balances[msg.sender] < _amount) {
            revert InsufficientBalance(balances[msg.sender], _amount);
        }

        balances[msg.sender] -= _amount;
        balances[_to] += _amount;
    }

    // 优势：更省 Gas，更清晰的错误信息
}
```

### 三种错误处理对比
```
┌──────────────────────────────────────────────────────────────┐
│ 方法      │ 使用场景              │ Gas 退还  │ Gas 消耗    │
├──────────────────────────────────────────────────────────────┤
│ require   │ 输入验证、条件检查    │ ✅ 是     │ 中等        │
│ assert    │ 内部错误、不变量      │ ❌ 否*    │ 高          │
│ revert    │ 复杂条件、自定义错误  │ ✅ 是     │ 低（自定义）│
└──────────────────────────────────────────────────────────────┘
* 0.8.0+ assert 也会退还 Gas
```

## 6.5 Try-Catch

处理外部调用的错误（0.6.0+）

```solidity
interface IExternal {
    function riskyFunction(uint256 value) external returns (uint256);
}

contract TryCatchExample {
    event SuccessLog(uint256 result);
    event ErrorLog(string reason);
    event LowLevelErrorLog(bytes data);

    IExternal public externalContract;

    constructor(address _external) {
        externalContract = IExternal(_external);
    }

    // Try-Catch 外部调用
    function callExternal(uint256 _value) public {
        try externalContract.riskyFunction(_value) returns (uint256 result) {
            // 成功
            emit SuccessLog(result);
        } catch Error(string memory reason) {
            // require/revert 的错误消息
            emit ErrorLog(reason);
        } catch (bytes memory lowLevelData) {
            // assert 或其他低级错误
            emit LowLevelErrorLog(lowLevelData);
        }
    }

    // Try-Catch 创建合约
    function createContract(uint256 _value) public {
        try new SomeContract(_value) returns (SomeContract instance) {
            // 合约创建成功
        } catch Error(string memory reason) {
            // 构造函数 revert
        } catch {
            // 其他错误
        }
    }

    // Try-Catch 外部函数调用（带返回值）
    function getExternalData() public returns (uint256) {
        try externalContract.riskyFunction(100) returns (uint256 result) {
            return result;
        } catch {
            return 0;  // 默认值
        }
    }
}

contract SomeContract {
    constructor(uint256 _value) {
        require(_value > 0, "Value must be positive");
    }
}
```

## 📝 本章小结

- ✅ 条件语句：if-else, 三元运算符
- ✅ 循环：for, while, do-while, break, continue
- ✅ 错误处理：require, assert, revert, 自定义错误

- ✅ Gas 优化技巧

## 🎯 最佳实践

1. **优先使用自定义错误**：节省 Gas
2. **require 用于输入验证**：用户输入、外部条件
3. **assert 用于不变量检查**：内部逻辑错误
4. **避免大循环**：限制迭代次数
5. **缓存数组长度**：循环中不重复读取
6. **使用 unchecked**：确定不会溢出时节省 Gas
