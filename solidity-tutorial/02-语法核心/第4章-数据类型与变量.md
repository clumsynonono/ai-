# 第4章：数据类型与变量

Solidity 是静态类型语言，变量类型在编译时确定。本章将详细介绍 Solidity 的各种数据类型。

## 4.1 值类型 (Value Types)

值类型变量在赋值或传参时会复制值。

### 布尔类型 (bool)
```solidity
contract BoolExample {
    bool public isActive = true;
    bool public isCompleted = false;

    // 布尔运算
    function boolOperations() public pure returns (bool, bool, bool) {
        bool a = true;
        bool b = false;

        return (
            !a,        // NOT: false
            a && b,    // AND: false
            a || b     // OR: true
        );
    }

    // 比较运算返回 bool
    function compare(uint256 x, uint256 y) public pure returns (bool) {
        return x > y;  // >, <, >=, <=, ==, !=
    }
}
```

### 整数类型
```solidity
contract IntegerTypes {
    // 无符号整数（只能是正数或零）
    uint8 public smallNumber;    // 0 到 2^8-1 (255)
    uint16 public mediumNumber;  // 0 到 2^16-1 (65535)
    uint256 public largeNumber;  // 0 到 2^256-1（默认推荐）
    uint public defaultUint;     // uint 等同于 uint256

    // 有符号整数（可以是负数）
    int8 public smallInt;        // -128 到 127
    int256 public largeInt;      // -2^255 到 2^255-1
    int public defaultInt;       // int 等同于 int256

    // 整数运算
    function mathOperations() public pure returns (uint256) {
        uint256 a = 10;
        uint256 b = 3;

        uint256 sum = a + b;         // 13
        uint256 difference = a - b;  // 7
        uint256 product = a * b;     // 30
        uint256 quotient = a / b;    // 3 (整数除法)
        uint256 remainder = a % b;   // 1 (取模)
        uint256 power = a ** 2;      // 100 (幂运算)

        return sum;
    }

    // 0.8.0+ 自动检查溢出
    function overflowExample() public pure returns (uint8) {
        uint8 max = 255;
        // return max + 1;  // ❌ 会 revert，不会溢出到 0
        return max;
    }

    // 使用 unchecked 禁用溢出检查（节省 Gas）
    function uncheckedExample() public pure returns (uint8) {
        uint8 max = 255;
        unchecked {
            return max + 1;  // 溢出到 0（不推荐）
        }
    }
}
```

### 地址类型 (address)
```solidity
contract AddressExample {
    address public owner;
    address payable public wallet;  // 可以接收 ETH

    constructor() {
        owner = msg.sender;
        wallet = payable(msg.sender);  // 转换为 payable
    }

    // 地址属性
    function getAddressInfo(address _addr) public view returns (
        uint256 balance,
        bytes32 codeHash
    ) {
        return (
            _addr.balance,          // 地址的 ETH 余额（wei）
            _addr.codehash         // 合约代码哈希（EOA 为特定值）
        );
    }

    // 地址比较
    function compareAddresses(address a, address b) public pure returns (bool) {
        return a == b;
    }

    // 零地址检查（常用）
    function isValidAddress(address _addr) public pure returns (bool) {
        return _addr != address(0);
    }

    // 转账 ETH（使用 payable 地址）
    function sendEther(address payable _to, uint256 _amount) public {
        require(address(this).balance >= _amount, "Insufficient balance");
        _to.transfer(_amount);  // 方法 1：transfer（失败会 revert）
        // (bool success, ) = _to.call{value: _amount}("");  // 方法 2：call（推荐）
        // require(success, "Transfer failed");
    }
}
```

### 字节类型
```solidity
contract BytesExample {
    // 固定大小字节数组
    bytes1 public singleByte;     // 1 字节
    bytes32 public hash;          // 32 字节（常用于哈希）

    // 动态字节数组
    bytes public dynamicBytes;

    function byteOperations() public pure returns (
        bytes1,
        bytes32
    ) {
        bytes1 b1 = 0x42;           // 单个字节
        bytes32 b32 = "Hello";       // 自动填充到 32 字节

        return (b1, b32);
    }

    // bytes vs string
    function bytesVsString() public pure returns (bytes memory, string memory) {
        bytes memory b = "Hello";    // 可以访问单个字节
        string memory s = "World";   // 不能直接访问单个字节

        // bytes 可以修改
        b[0] = 0x48;  // "Hello" → "Hello"（修改首字母）

        return (b, s);
    }
}
```

### 枚举类型 (enum)
```solidity
contract EnumExample {
    // 定义枚举
    enum Status {
        Pending,    // 0
        Approved,   // 1
        Rejected,   // 2
        Canceled    // 3
    }

    Status public currentStatus;

    function setStatus(Status _status) public {
        currentStatus = _status;
    }

    function approve() public {
        currentStatus = Status.Approved;
    }

    function getStatus() public view returns (Status) {
        return currentStatus;
    }

    // 枚举与 uint 转换
    function enumToUint() public view returns (uint) {
        return uint(currentStatus);
    }

    // 获取枚举最小和最大值
    function getMinMax() public pure returns (Status, Status) {
        return (
            Status.min,  // Pending (0)
            Status.max   // Canceled (3)
        );
    }
}
```

## 4.2 引用类型 (Reference Types)

引用类型传递引用而非值，需要指定数据位置。

### 数组 (Array)
```solidity
contract ArrayExample {
    // 固定大小数组
    uint256[5] public fixedArray;

    // 动态数组
    uint256[] public dynamicArray;
    address[] public users;

    // 初始化数组
    constructor() {
        fixedArray = [1, 2, 3, 4, 5];
    }

    // 数组操作
    function arrayOperations() public {
        // 添加元素
        dynamicArray.push(1);
        dynamicArray.push(2);

        // 删除最后一个元素
        dynamicArray.pop();

        // 访问元素
        uint256 first = dynamicArray[0];

        // 修改元素
        dynamicArray[0] = 100;

        // 获取长度
        uint256 len = dynamicArray.length;

        // 删除元素（设为默认值，但不改变长度）
        delete dynamicArray[0];  // 设为 0
    }

    // 返回整个数组（小心 Gas 消耗）
    function getArray() public view returns (uint256[] memory) {
        return dynamicArray;
    }

    // 内存中创建数组
    function createMemoryArray() public pure returns (uint256[] memory) {
        uint256[] memory arr = new uint256[](5);  // 必须指定大小
        arr[0] = 1;
        arr[1] = 2;
        return arr;
    }

    // 多维数组
    uint256[][] public matrix;

    function addRow() public {
        uint256[] memory row = new uint256[](3);
        row[0] = 1;
        row[1] = 2;
        row[2] = 3;
        matrix.push(row);
    }
}
```

### 结构体 (Struct)
```solidity
contract StructExample {
    // 定义结构体
    struct User {
        string name;
        uint256 age;
        address wallet;
        bool isActive;
    }

    // 存储结构体
    User public admin;
    User[] public users;
    mapping(address => User) public userMap;

    // 创建结构体的多种方式
    function createUser() public {
        // 方式 1：按顺序赋值
        User memory user1 = User("Alice", 25, msg.sender, true);

        // 方式 2：键值对赋值（推荐）
        User memory user2 = User({
            name: "Bob",
            age: 30,
            wallet: msg.sender,
            isActive: true
        });

        // 方式 3：分步赋值
        User memory user3;
        user3.name = "Charlie";
        user3.age = 28;
        user3.wallet = msg.sender;
        user3.isActive = true;

        // 添加到数组
        users.push(user1);
        users.push(user2);
        users.push(user3);
    }

    // 修改结构体
    function updateUser(uint256 index, string memory newName) public {
        require(index < users.length, "Invalid index");
        users[index].name = newName;
    }

    // 从 mapping 中读取
    function getUser(address _addr) public view returns (User memory) {
        return userMap[_addr];
    }

    // 嵌套结构体
    struct Company {
        string name;
        User[] employees;
    }
}
```

### 映射 (Mapping)
```solidity
contract MappingExample {
    // 基本映射
    mapping(address => uint256) public balances;
    mapping(address => bool) public isRegistered;
    mapping(uint256 => string) public idToName;

    // 嵌套映射
    mapping(address => mapping(address => uint256)) public allowances;

    // 映射操作
    function mappingOperations() public {
        // 设置值
        balances[msg.sender] = 100;

        // 读取值（不存在返回默认值）
        uint256 balance = balances[msg.sender];

        // 更新值
        balances[msg.sender] += 50;

        // 删除值（重置为默认值）
        delete balances[msg.sender];  // 重置为 0
    }

    // 检查键是否存在（技巧）
    mapping(address => bool) private _exists;

    function addUser(address _user) public {
        require(!_exists[_user], "User already exists");
        _exists[_user] = true;
        balances[_user] = 0;
    }

    // 嵌套映射示例：授权
    function approve(address _spender, uint256 _amount) public {
        allowances[msg.sender][_spender] = _amount;
    }

    function getAllowance(address _owner, address _spender)
        public
        view
        returns (uint256)
    {
        return allowances[_owner][_spender];
    }
}
```

## 4.3 数据位置 (Data Location)

引用类型必须指定数据位置：`storage`、`memory` 或 `calldata`。

```solidity
contract DataLocationExample {
    uint256[] public storageArray;  // storage（状态变量）

    struct Person {
        string name;
        uint256 age;
    }

    Person[] public people;

    // storage：永久存储在区块链上
    function storageExample() public {
        Person storage person = people[0];  // 引用
        person.name = "Alice";  // 修改会影响原数据
    }

    // memory：临时存储，函数执行后消失
    function memoryExample(uint256[] memory _arr) public pure returns (uint256) {
        // _arr 是临时副本
        uint256[] memory tempArr = new uint256[](5);
        tempArr[0] = 100;
        return tempArr[0];
    }

    // calldata：只读的外部函数参数（最省 Gas）
    function calldataExample(uint256[] calldata _arr) external pure returns (uint256) {
        // _arr 不可修改
        // _arr[0] = 100;  // ❌ 错误
        return _arr[0];  // ✅ 只读
    }

    // 数据位置的赋值行为
    function assignmentBehavior() public {
        // storage -> storage：创建引用
        uint256[] storage arr1 = storageArray;
        arr1.push(1);  // storageArray 也会改变

        // memory -> memory：创建引用
        uint256[] memory arr2 = new uint256[](3);
        uint256[] memory arr3 = arr2;  // 引用
        arr3[0] = 100;  // arr2[0] 也变成 100

        // storage -> memory：复制
        uint256[] memory arr4 = storageArray;
        arr4[0] = 999;  // storageArray 不受影响
    }
}
```

## 4.4 常量与不可变量

### constant（编译时常量）
```solidity
contract ConstantExample {
    // 必须在声明时赋值
    uint256 public constant MAX_SUPPLY = 1000000;
    address public constant OWNER = 0x5B38Da6a701c568545dCfcB03FcB875f56beddC4;
    string public constant NAME = "MyToken";

    // 节省 Gas：直接嵌入到字节码中
    function getMaxSupply() public pure returns (uint256) {
        return MAX_SUPPLY;  // 非常便宜
    }
}
```

### immutable（部署时常量）
```solidity
contract ImmutableExample {
    // 可以在构造函数中赋值
    address public immutable owner;
    uint256 public immutable deployTime;
    uint256 public immutable maxSupply;

    constructor(uint256 _maxSupply) {
        owner = msg.sender;
        deployTime = block.timestamp;
        maxSupply = _maxSupply;
    }

    // immutable 比普通状态变量省 Gas
    function getOwner() public view returns (address) {
        return owner;
    }
}
```

### constant vs immutable vs 状态变量
```solidity
contract ComparisonExample {
    uint256 public constant CONST_VAR = 100;      // 最便宜
    uint256 public immutable IMMUTABLE_VAR;       // 较便宜
    uint256 public stateVar;                      // 最贵

    constructor() {
        IMMUTABLE_VAR = 200;
        stateVar = 300;
    }

    // Gas 对比
    function readConstant() public pure returns (uint256) {
        return CONST_VAR;  // ~100 gas
    }

    function readImmutable() public view returns (uint256) {
        return IMMUTABLE_VAR;  // ~200 gas
    }

    function readState() public view returns (uint256) {
        return stateVar;  // ~2100 gas
    }
}
```

## 4.5 默认值

所有变量都有默认值，不存在 `null` 或 `undefined`。

```solidity
contract DefaultValues {
    bool public defaultBool;           // false
    uint256 public defaultUint;        // 0
    int256 public defaultInt;          // 0
    address public defaultAddress;     // 0x0000000000000000000000000000000000000000
    bytes32 public defaultBytes32;     // 0x0000...0000
    string public defaultString;       // ""
    uint256[] public defaultArray;     // [] (空数组)

    enum Status { Pending, Approved }
    Status public defaultStatus;       // Status.Pending (0)

    struct User {
        string name;
        uint256 age;
    }
    User public defaultUser;           // {name: "", age: 0}
}
```

## 📝 本章小结

- ✅ 值类型：bool, int, uint, address, bytes, enum
- ✅ 引用类型：array, struct, mapping
- ✅ 数据位置：storage, memory, calldata
- ✅ 常量：constant, immutable
- ✅ 默认值和类型转换

## 🎯 重要提示

1. **uint256 vs uint8**：除非需要打包，否则使用 uint256（EVM 原生 256 位）
2. **address vs address payable**：需要转账时使用 payable
3. **memory vs calldata**：外部函数参数优先使用 calldata
4. **mapping 无法遍历**：需要遍历时配合数组使用
5. **constant 和 immutable 节省 Gas**：尽量使用
