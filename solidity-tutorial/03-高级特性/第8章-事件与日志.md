# 第8章：事件与日志

事件（Events）是 Solidity 中记录链上活动的重要机制，也是智能合约与前端交互的主要方式。

## 8.1 事件基础

### 定义和触发事件
```solidity
contract EventBasics {
    // 定义事件
    event Transfer(address from, address to, uint256 amount);
    event Deposit(address indexed user, uint256 amount);
    event Withdrawal(address indexed user, uint256 amount, uint256 timestamp);

    mapping(address => uint256) public balances;

    // 触发事件
    function deposit() public payable {
        balances[msg.sender] += msg.value;

        // 使用 emit 触发事件
        emit Deposit(msg.sender, msg.value);
    }

    function transfer(address to, uint256 amount) public {
        require(balances[msg.sender] >= amount, "Insufficient balance");

        balances[msg.sender] -= amount;
        balances[to] += amount;

        emit Transfer(msg.sender, to, amount);
    }

    function withdraw(uint256 amount) public {
        require(balances[msg.sender] >= amount, "Insufficient balance");

        balances[msg.sender] -= amount;
        payable(msg.sender).transfer(amount);

        emit Withdrawal(msg.sender, amount, block.timestamp);
    }
}
```

## 8.2 indexed 参数

使用 `indexed` 关键字可以让事件参数可搜索。

```solidity
contract IndexedEvents {
    // indexed 参数可以用于过滤
    event Transfer(
        address indexed from,    // 可索引
        address indexed to,      // 可索引
        uint256 amount           // 不可索引
    );

    event OrderCreated(
        uint256 indexed orderId,     // 可索引
        address indexed buyer,       // 可索引
        address indexed seller,      // 可索引
        uint256 price                // 不可索引（最多 3 个 indexed）
    );

    // ⚠️ 最多 3 个 indexed 参数
    event TooManyIndexed(
        address indexed a,
        address indexed b,
        address indexed c
        // uint256 indexed d  // ❌ 错误：超过 3 个
    );

    function makeTransfer(address to, uint256 amount) public {
        emit Transfer(msg.sender, to, amount);
    }
}
```

### indexed vs 非 indexed
```
┌──────────────────────────────────────────────────────────┐
│ 特性          │ indexed              │ 非 indexed       │
├──────────────────────────────────────────────────────────┤
│ 可搜索性      │ ✅ 可以通过值过滤    │ ❌ 不能过滤      │
│ 存储位置      │ 存储在 topics 中     │ 存储在 data 中   │
│ Gas 成本      │ 更高                 │ 更低             │
│ 数据完整性    │ 大数据会被哈希       │ 完整存储         │
│ 数量限制      │ 最多 3 个            │ 无限制           │
└──────────────────────────────────────────────────────────┘
```

## 8.3 匿名事件

匿名事件不存储事件签名，可以节省一个 topic 槽位。

```solidity
contract AnonymousEvents {
    // 普通事件：占用 1 个 topic（事件签名）+ 3 个 indexed 参数
    event NormalEvent(
        address indexed a,
        address indexed b,
        address indexed c,
        uint256 value
    );

    // 匿名事件：不占用事件签名的 topic，可以有 4 个 indexed
    event AnonymousEvent(
        address indexed a,
        address indexed b,
        address indexed c,
        uint256 indexed d
    ) anonymous;

    function emitEvents() public {
        emit NormalEvent(msg.sender, msg.sender, msg.sender, 100);
        emit AnonymousEvent(msg.sender, msg.sender, msg.sender, 200);
    }

    // ⚠️ 匿名事件难以区分，不推荐频繁使用
}
```

## 8.4 事件的实际应用

### 示例 1：代币转账记录
```solidity
contract Token {
    string public name = "MyToken";
    mapping(address => uint256) public balances;

    event Transfer(address indexed from, address indexed to, uint256 value);
    event Approval(address indexed owner, address indexed spender, uint256 value);
    event Mint(address indexed to, uint256 amount);
    event Burn(address indexed from, uint256 amount);

    function transfer(address to, uint256 amount) public returns (bool) {
        require(balances[msg.sender] >= amount, "Insufficient balance");

        balances[msg.sender] -= amount;
        balances[to] += amount;

        emit Transfer(msg.sender, to, amount);
        return true;
    }

    function mint(address to, uint256 amount) public {
        balances[to] += amount;
        emit Mint(to, amount);
    }

    function burn(uint256 amount) public {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        balances[msg.sender] -= amount;
        emit Burn(msg.sender, amount);
    }
}
```

### 示例 2：订单系统
```solidity
contract OrderBook {
    enum OrderStatus { Created, Filled, Cancelled }

    struct Order {
        address buyer;
        address seller;
        uint256 price;
        OrderStatus status;
    }

    mapping(uint256 => Order) public orders;
    uint256 public nextOrderId;

    event OrderCreated(uint256 indexed orderId, address indexed buyer, uint256 price);
    event OrderFilled(uint256 indexed orderId, address indexed seller);
    event OrderCancelled(uint256 indexed orderId);
    event OrderStatusChanged(uint256 indexed orderId, OrderStatus newStatus);

    function createOrder(uint256 price) public payable {
        require(msg.value == price, "Incorrect payment");

        uint256 orderId = nextOrderId++;
        orders[orderId] = Order({
            buyer: msg.sender,
            seller: address(0),
            price: price,
            status: OrderStatus.Created
        });

        emit OrderCreated(orderId, msg.sender, price);
    }

    function fillOrder(uint256 orderId) public {
        Order storage order = orders[orderId];
        require(order.status == OrderStatus.Created, "Order not available");

        order.seller = msg.sender;
        order.status = OrderStatus.Filled;

        payable(msg.sender).transfer(order.price);

        emit OrderFilled(orderId, msg.sender);
        emit OrderStatusChanged(orderId, OrderStatus.Filled);
    }

    function cancelOrder(uint256 orderId) public {
        Order storage order = orders[orderId];
        require(order.buyer == msg.sender, "Not the buyer");
        require(order.status == OrderStatus.Created, "Cannot cancel");

        order.status = OrderStatus.Cancelled;
        payable(order.buyer).transfer(order.price);

        emit OrderCancelled(orderId);
        emit OrderStatusChanged(orderId, OrderStatus.Cancelled);
    }
}
```

### 示例 3：权限变更日志
```solidity
contract AccessControl {
    mapping(address => bool) public admins;
    mapping(address => mapping(bytes32 => bool)) public permissions;

    event AdminAdded(address indexed admin, address indexed addedBy);
    event AdminRemoved(address indexed admin, address indexed removedBy);
    event PermissionGranted(address indexed user, bytes32 indexed permission, address indexed grantedBy);
    event PermissionRevoked(address indexed user, bytes32 indexed permission, address indexed revokedBy);

    modifier onlyAdmin() {
        require(admins[msg.sender], "Not an admin");
        _;
    }

    constructor() {
        admins[msg.sender] = true;
        emit AdminAdded(msg.sender, msg.sender);
    }

    function addAdmin(address newAdmin) public onlyAdmin {
        admins[newAdmin] = true;
        emit AdminAdded(newAdmin, msg.sender);
    }

    function removeAdmin(address admin) public onlyAdmin {
        admins[admin] = false;
        emit AdminRemoved(admin, msg.sender);
    }

    function grantPermission(address user, bytes32 permission) public onlyAdmin {
        permissions[user][permission] = true;
        emit PermissionGranted(user, permission, msg.sender);
    }

    function revokePermission(address user, bytes32 permission) public onlyAdmin {
        permissions[user][permission] = false;
        emit PermissionRevoked(user, permission, msg.sender);
    }
}
```

## 8.5 从前端监听事件

### JavaScript (ethers.js) 示例
```javascript
// 连接到合约
const contract = new ethers.Contract(contractAddress, abi, provider);

// 1. 监听单个事件
contract.on("Transfer", (from, to, amount, event) => {
    console.log(`Transfer from ${from} to ${to}: ${amount}`);
    console.log("Transaction hash:", event.transactionHash);
});

// 2. 过滤特定地址的事件
const filter = contract.filters.Transfer(myAddress, null);
contract.on(filter, (from, to, amount) => {
    console.log(`Received ${amount} from ${from}`);
});

// 3. 查询历史事件
const events = await contract.queryFilter(
    contract.filters.Transfer(),
    fromBlock,
    toBlock
);

events.forEach(event => {
    console.log(event.args);
});

// 4. 一次性监听
contract.once("Deposit", (user, amount) => {
    console.log(`Deposit detected: ${user} deposited ${amount}`);
});

// 5. 移除监听器
contract.removeAllListeners("Transfer");
```

## 8.6 Gas 优化

### 事件 vs 存储
```solidity
contract GasComparison {
    // 存储历史（昂贵）
    struct Transaction {
        address from;
        address to;
        uint256 amount;
        uint256 timestamp;
    }

    Transaction[] public transactionHistory;  // 存储成本高

    event TransferEvent(
        address indexed from,
        address indexed to,
        uint256 amount,
        uint256 timestamp
    );  // 事件成本低

    // ❌ 昂贵：写入存储
    function recordInStorage(address to, uint256 amount) public {
        transactionHistory.push(Transaction({
            from: msg.sender,
            to: to,
            amount: amount,
            timestamp: block.timestamp
        }));
        // 高 Gas 消耗
    }

    // ✅ 便宜：使用事件
    function recordInEvent(address to, uint256 amount) public {
        emit TransferEvent(msg.sender, to, amount, block.timestamp);
        // 低 Gas 消耗
    }

    /*
    Gas 对比（大约）：
    - 存储：~20,000 gas per slot
    - 事件：~375 gas + ~375 per indexed topic + ~8 gas per byte
    */
}
```

### 优化事件的技巧
```solidity
contract EventOptimization {
    // ✅ 好：使用 indexed 用于常用过滤字段
    event Transfer(address indexed from, address indexed to, uint256 amount);

    // ❌ 差：所有参数都 indexed（浪费）
    event BadEvent(
        uint256 indexed a,
        uint256 indexed b,
        uint256 indexed c,
        uint256 indexed d  // ❌ 超过 3 个
    );

    // ✅ 好：合理使用 indexed
    event GoodEvent(
        address indexed user,      // 常用于过滤
        uint256 indexed action,    // 常用于过滤
        bytes data                 // 不常过滤，不使用 indexed
    );

    // 💡 技巧：对于大数据，不要使用 indexed
    // indexed 的大数据会被哈希，丢失原始信息
    event DataEvent(
        address indexed user,
        string message  // 不 indexed，保留完整信息
    );
}
```

## 8.7 调试技巧

### 使用事件调试
```solidity
contract Debugging {
    event Debug(string message);
    event DebugAddress(string label, address addr);
    event DebugUint(string label, uint256 value);
    event DebugBool(string label, bool value);

    function complexFunction(uint256 x) public {
        emit Debug("Starting complexFunction");
        emit DebugUint("Input x", x);

        uint256 y = x * 2;
        emit DebugUint("Computed y", y);

        require(y > 10, "Y too small");
        emit Debug("Passed require check");

        if (y > 100) {
            emit Debug("Y is greater than 100");
        } else {
            emit Debug("Y is less than or equal to 100");
        }

        emit Debug("complexFunction completed");
    }

    // 在开发环境中使用，生产环境删除
}
```

## 🎯 最佳实践

1. **合理使用 indexed**：常用于过滤的字段才 indexed
2. **事件优于存储**：历史记录用事件，节省 Gas
3. **一致的命名**：遵循标准（如 ERC20 的 Transfer）
4. **详细的事件信息**：便于前端展示和调试
5. **考虑隐私**：敏感数据不要放在事件中（链上公开）
