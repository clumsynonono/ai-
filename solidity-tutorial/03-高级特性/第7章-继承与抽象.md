# 第7章：继承与抽象

Solidity 支持面向对象编程，包括继承、多态等特性。本章学习如何编写可复用和可扩展的合约。

## 7.1 基础继承

### 单继承
```solidity
// 父合约
contract Animal {
    string public name;
    uint256 public age;

    constructor(string memory _name, uint256 _age) {
        name = _name;
        age = _age;
    }

    function eat() public virtual returns (string memory) {
        return "Animal is eating";
    }

    function sleep() public pure returns (string memory) {
        return "Animal is sleeping";
    }
}

// 子合约继承父合约
contract Dog is Animal {
    string public breed;

    // 调用父合约构造函数
    constructor(string memory _name, uint256 _age, string memory _breed)
        Animal(_name, _age)
    {
        breed = _breed;
    }

    // 重写父合约函数
    function eat() public pure override returns (string memory) {
        return "Dog is eating bones";
    }

    // 子合约的新函数
    function bark() public pure returns (string memory) {
        return "Woof!";
    }
}
```

### 访问父合约
```solidity
contract Parent {
    uint256 public value = 10;

    function getValue() public view virtual returns (uint256) {
        return value;
    }
}

contract Child is Parent {
    // 使用 super 调用父合约
    function getParentValue() public view returns (uint256) {
        return super.getValue();
    }

    // 直接使用父合约名称
    function getParentValueDirect() public view returns (uint256) {
        return Parent.getValue();
    }

    // 重写并调用父合约
    function getValue() public view override returns (uint256) {
        uint256 parentValue = super.getValue();
        return parentValue * 2;  // 父值的两倍
    }
}
```

## 7.2 多重继承

Solidity 支持多重继承，但需要注意继承顺序。

```solidity
contract A {
    function foo() public pure virtual returns (string memory) {
        return "A";
    }

    function bar() public pure virtual returns (string memory) {
        return "A";
    }
}

contract B {
    function foo() public pure virtual returns (string memory) {
        return "B";
    }

    function baz() public pure virtual returns (string memory) {
        return "B";
    }
}

// 多重继承：从最基础到最派生
contract C is A, B {
    // 必须重写冲突的函数
    function foo() public pure override(A, B) returns (string memory) {
        return "C";
    }

    // 可以调用两个父合约
    function getAllFoo() public pure returns (string memory, string memory, string memory) {
        return (A.foo(), B.foo(), foo());
    }
}
```

### 继承顺序（线性化）
```solidity
//     A
//    / \
//   B   C
//    \ /
//     D

contract A {
    function foo() public pure virtual returns (string memory) {
        return "A";
    }
}

contract B is A {
    function foo() public pure override virtual returns (string memory) {
        return "B";
    }
}

contract C is A {
    function foo() public pure override virtual returns (string memory) {
        return "C";
    }
}

// ✅ 正确：从最基础到最派生
contract D is B, C {
    function foo() public pure override(B, C) returns (string memory) {
        return super.foo();  // 会调用 C.foo()
    }
}

// ❌ 错误：顺序不对
// contract D is C, B { }  // 编译错误
```

### 多重继承中的构造函数
```solidity
contract Base1 {
    uint256 public x;

    constructor(uint256 _x) {
        x = _x;
    }
}

contract Base2 {
    uint256 public y;

    constructor(uint256 _y) {
        y = _y;
    }
}

// 方式 1：在继承列表中初始化
contract Derived1 is Base1(10), Base2(20) {
    // x = 10, y = 20
}

// 方式 2：在构造函数中初始化
contract Derived2 is Base1, Base2 {
    constructor(uint256 _x, uint256 _y) Base1(_x) Base2(_y) {
        // 动态传参
    }
}

// 方式 3：混合
contract Derived3 is Base1(10), Base2 {
    constructor(uint256 _y) Base2(_y) {
        // Base1 固定，Base2 动态
    }
}
```

## 7.3 虚函数与重写

### virtual 和 override
```solidity
contract Base {
    // virtual：可以被子合约重写
    function regularFunction() public pure virtual returns (string memory) {
        return "Base";
    }

    // 没有 virtual：不能被重写
    function finalFunction() public pure returns (string memory) {
        return "Final";
    }

    // 虚函数可以有实现
    function withImplementation() public pure virtual returns (uint256) {
        return 42;
    }
}

contract Derived is Base {
    // override：重写父合约函数
    function regularFunction() public pure override returns (string memory) {
        return "Derived";
    }

    // ❌ 错误：没有 virtual，不能重写
    // function finalFunction() public pure override returns (string memory) { }

    // 重写并保持 virtual（允许进一步重写）
    function withImplementation()
        public
        pure
        virtual
        override
        returns (uint256)
    {
        return 100;
    }
}

contract DerivedDerived is Derived {
    // 可以继续重写
    function withImplementation()
        public
        pure
        override
        returns (uint256)
    {
        return 200;
    }
}
```

### 状态变量的重写（0.6.0+）
```solidity
contract Parent {
    uint256 public value = 10;

    // public 状态变量自动生成 getter，也可以重写
    function value() public view virtual returns (uint256) {
        return 10;
    }
}

contract Child is Parent {
    // 重写 getter
    function value() public pure override returns (uint256) {
        return 20;
    }
}
```

## 7.4 抽象合约

抽象合约包含至少一个未实现的函数。

```solidity
// 抽象合约不能直接部署
abstract contract AbstractToken {
    string public name;

    constructor(string memory _name) {
        name = _name;
    }

    // 未实现的函数（没有函数体）
    function transfer(address to, uint256 amount) public virtual returns (bool);

    // 可以有已实现的函数
    function getName() public view returns (string memory) {
        return name;
    }

    // 可以有抽象函数
    function totalSupply() public view virtual returns (uint256);
}

// 实现抽象合约
contract MyToken is AbstractToken {
    uint256 private _totalSupply;
    mapping(address => uint256) public balances;

    constructor(string memory _name, uint256 _supply)
        AbstractToken(_name)
    {
        _totalSupply = _supply;
        balances[msg.sender] = _supply;
    }

    // 必须实现所有抽象函数
    function transfer(address to, uint256 amount)
        public
        override
        returns (bool)
    {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        balances[msg.sender] -= amount;
        balances[to] += amount;
        return true;
    }

    function totalSupply() public view override returns (uint256) {
        return _totalSupply;
    }
}
```

## 7.5 接口

接口定义合约的外部接口，用于与其他合约交互。

```solidity
// 接口特点：
// - 只能有函数声明，不能有实现
// - 所有函数必须是 external
// - 不能有构造函数
// - 不能有状态变量
// - 可以继承其他接口

interface IERC20 {
    // 事件
    event Transfer(address indexed from, address indexed to, uint256 value);
    event Approval(address indexed owner, address indexed spender, uint256 value);

    // 函数声明（必须是 external）
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address to, uint256 amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function approve(address spender, uint256 amount) external returns (bool);
    function transferFrom(address from, address to, uint256 amount) external returns (bool);
}

// 实现接口
contract MyERC20 is IERC20 {
    string public name = "MyToken";
    uint256 private _totalSupply;
    mapping(address => uint256) private _balances;
    mapping(address => mapping(address => uint256)) private _allowances;

    constructor(uint256 supply) {
        _totalSupply = supply;
        _balances[msg.sender] = supply;
    }

    // 实现所有接口函数
    function totalSupply() external view override returns (uint256) {
        return _totalSupply;
    }

    function balanceOf(address account) external view override returns (uint256) {
        return _balances[account];
    }

    function transfer(address to, uint256 amount) external override returns (bool) {
        _transfer(msg.sender, to, amount);
        return true;
    }

    function allowance(address owner, address spender)
        external
        view
        override
        returns (uint256)
    {
        return _allowances[owner][spender];
    }

    function approve(address spender, uint256 amount)
        external
        override
        returns (bool)
    {
        _allowances[msg.sender][spender] = amount;
        emit Approval(msg.sender, spender, amount);
        return true;
    }

    function transferFrom(address from, address to, uint256 amount)
        external
        override
        returns (bool)
    {
        require(_allowances[from][msg.sender] >= amount, "Insufficient allowance");
        _allowances[from][msg.sender] -= amount;
        _transfer(from, to, amount);
        return true;
    }

    function _transfer(address from, address to, uint256 amount) internal {
        require(_balances[from] >= amount, "Insufficient balance");
        _balances[from] -= amount;
        _balances[to] += amount;
        emit Transfer(from, to, amount);
    }
}

// 使用接口与其他合约交互
contract TokenUser {
    IERC20 public token;

    constructor(address tokenAddress) {
        token = IERC20(tokenAddress);  // 通过接口连接
    }

    function checkBalance(address account) public view returns (uint256) {
        return token.balanceOf(account);
    }

    function transferTokens(address to, uint256 amount) public {
        token.transfer(to, amount);
    }
}
```

## 7.6 抽象合约 vs 接口

```
┌─────────────────────────────────────────────────────────┐
│ 特性              │ 抽象合约           │ 接口           │
├─────────────────────────────────────────────────────────┤
│ 函数实现          │ ✅ 部分可以        │ ❌ 都不可以    │
│ 状态变量          │ ✅ 可以            │ ❌ 不可以      │
│ 构造函数          │ ✅ 可以            │ ❌ 不可以      │
│ 函数可见性        │ 任意               │ 只能 external  │
│ 继承其他合约      │ ✅ 可以            │ 只能接口       │
│ 使用场景          │ 提供部分实现       │ 定义标准规范   │
└─────────────────────────────────────────────────────────┘
```

## 7.7 实战示例：权限管理系统

```solidity
// 抽象基础合约
abstract contract Ownable {
    address private _owner;

    event OwnershipTransferred(address indexed previousOwner, address indexed newOwner);

    constructor() {
        _owner = msg.sender;
        emit OwnershipTransferred(address(0), _owner);
    }

    modifier onlyOwner() {
        require(owner() == msg.sender, "Ownable: caller is not the owner");
        _;
    }

    function owner() public view virtual returns (address) {
        return _owner;
    }

    function transferOwnership(address newOwner) public virtual onlyOwner {
        require(newOwner != address(0), "Ownable: new owner is the zero address");
        emit OwnershipTransferred(_owner, newOwner);
        _owner = newOwner;
    }

    function renounceOwnership() public virtual onlyOwner {
        emit OwnershipTransferred(_owner, address(0));
        _owner = address(0);
    }
}

// 暂停功能抽象合约
abstract contract Pausable is Ownable {
    bool private _paused;

    event Paused(address account);
    event Unpaused(address account);

    constructor() {
        _paused = false;
    }

    modifier whenNotPaused() {
        require(!paused(), "Pausable: paused");
        _;
    }

    modifier whenPaused() {
        require(paused(), "Pausable: not paused");
        _;
    }

    function paused() public view virtual returns (bool) {
        return _paused;
    }

    function pause() public virtual onlyOwner whenNotPaused {
        _paused = true;
        emit Paused(msg.sender);
    }

    function unpause() public virtual onlyOwner whenPaused {
        _paused = false;
        emit Unpaused(msg.sender);
    }
}

// 实际合约继承多个抽象合约
contract SecureVault is Pausable {
    mapping(address => uint256) public balances;

    function deposit() public payable whenNotPaused {
        balances[msg.sender] += msg.value;
    }

    function withdraw(uint256 amount) public whenNotPaused {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        balances[msg.sender] -= amount;
        payable(msg.sender).transfer(amount);
    }

    // Owner 可以紧急暂停合约
    function emergencyPause() external onlyOwner {
        pause();
    }
}
```

## 📝 本章小结

- ✅ 单继承和多重继承
- ✅ virtual 和 override 关键字
- ✅ 抽象合约：部分实现的基础合约
- ✅ 接口：纯定义，用于标准化
- ✅ 继承顺序和线性化
- ✅ super 和父合约调用

## 🎯 最佳实践

1. **优先使用接口**：定义标准（如 ERC20）
2. **抽象合约提供基础功能**：如 Ownable
3. **注意继承顺序**：从基础到派生
4. **适当使用 virtual**：需要扩展时才标记
5. **OpenZeppelin 合约库**：使用经过审计的合约
