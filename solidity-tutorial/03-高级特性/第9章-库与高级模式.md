# 第9章：库与高级模式

本章学习如何使用库（Library）复用代码，以及智能合约中的常见设计模式。

## 9.1 库（Library）基础

库是可复用的代码块，不能有状态变量，不能接收 ETH。

### 创建和使用库
```solidity
// 定义库
library MathLib {
    // 库函数通常是 internal 或 public
    function add(uint256 a, uint256 b) internal pure returns (uint256) {
        return a + b;
    }

    function sub(uint256 a, uint256 b) internal pure returns (uint256) {
        require(b <= a, "Subtraction overflow");
        return a - b;
    }

    function mul(uint256 a, uint256 b) internal pure returns (uint256) {
        return a * b;
    }

    function div(uint256 a, uint256 b) internal pure returns (uint256) {
        require(b > 0, "Division by zero");
        return a / b;
    }
}

// 使用库
contract Calculator {
    // 方式 1：直接调用
    function calculate(uint256 a, uint256 b) public pure returns (uint256) {
        return MathLib.add(a, b);
    }

    // 方式 2：使用 using for
    using MathLib for uint256;

    function calculate2(uint256 a, uint256 b) public pure returns (uint256) {
        return a.add(b);  // 更优雅的语法
    }
}
```

## 9.2 Using For 语法

`using A for B` 可以将库 A 的函数附加到类型 B 上。

```solidity
library StringLib {
    function length(string memory str) internal pure returns (uint256) {
        return bytes(str).length;
    }

    function concat(string memory a, string memory b)
        internal
        pure
        returns (string memory)
    {
        return string(abi.encodePacked(a, b));
    }

    function equals(string memory a, string memory b)
        internal
        pure
        returns (bool)
    {
        return keccak256(abi.encodePacked(a)) == keccak256(abi.encodePacked(b));
    }
}

contract StringUser {
    // 将 StringLib 附加到 string 类型
    using StringLib for string;

    function processString(string memory str) public pure returns (uint256) {
        // 可以像方法一样调用
        return str.length();
    }

    function combineStrings(string memory a, string memory b)
        public
        pure
        returns (string memory)
    {
        return a.concat(b);
    }

    function compareStrings(string memory a, string memory b)
        public
        pure
        returns (bool)
    {
        return a.equals(b);
    }
}
```

## 9.3 OpenZeppelin 库示例

### SafeMath（0.8.0 前必用）
```solidity
// 0.8.0 前需要 SafeMath 防止溢出
library SafeMath {
    function add(uint256 a, uint256 b) internal pure returns (uint256) {
        uint256 c = a + b;
        require(c >= a, "SafeMath: addition overflow");
        return c;
    }

    function sub(uint256 a, uint256 b) internal pure returns (uint256) {
        require(b <= a, "SafeMath: subtraction overflow");
        return a - b;
    }

    function mul(uint256 a, uint256 b) internal pure returns (uint256) {
        if (a == 0) return 0;
        uint256 c = a * b;
        require(c / a == b, "SafeMath: multiplication overflow");
        return c;
    }

    function div(uint256 a, uint256 b) internal pure returns (uint256) {
        require(b > 0, "SafeMath: division by zero");
        return a / b;
    }
}

// 0.8.0+ 不再需要，已内置溢出检查
```

### Address 库
```solidity
library Address {
    // 检查是否是合约地址
    function isContract(address account) internal view returns (bool) {
        uint256 size;
        assembly {
            size := extcodesize(account)
        }
        return size > 0;
    }

    // 安全转账
    function sendValue(address payable recipient, uint256 amount) internal {
        require(address(this).balance >= amount, "Address: insufficient balance");

        (bool success, ) = recipient.call{value: amount}("");
        require(success, "Address: unable to send value");
    }

    // 安全调用
    function functionCall(address target, bytes memory data)
        internal
        returns (bytes memory)
    {
        require(isContract(target), "Address: call to non-contract");

        (bool success, bytes memory returndata) = target.call(data);
        require(success, "Address: low-level call failed");

        return returndata;
    }
}

contract UsingAddress {
    using Address for address;
    using Address for address payable;

    function transfer(address payable to, uint256 amount) public {
        to.sendValue(amount);
    }

    function isUserContract(address user) public view returns (bool) {
        return user.isContract();
    }
}
```

## 9.4 常见设计模式

### 1. 工厂模式
```solidity
// 子合约
contract Token {
    string public name;
    address public owner;

    constructor(string memory _name, address _owner) {
        name = _name;
        owner = _owner;
    }
}

// 工厂合约
contract TokenFactory {
    Token[] public tokens;
    mapping(address => Token[]) public ownerTokens;

    event TokenCreated(address tokenAddress, string name, address owner);

    function createToken(string memory name) public returns (address) {
        Token newToken = new Token(name, msg.sender);
        tokens.push(newToken);
        ownerTokens[msg.sender].push(newToken);

        emit TokenCreated(address(newToken), name, msg.sender);
        return address(newToken);
    }

    function getTokenCount() public view returns (uint256) {
        return tokens.length;
    }

    function getUserTokens(address user) public view returns (Token[] memory) {
        return ownerTokens[user];
    }
}
```

### 2. 代理模式（可升级合约）
```solidity
// 逻辑合约
contract LogicV1 {
    uint256 public value;

    function setValue(uint256 _value) public {
        value = _value;
    }

    function getValue() public view returns (uint256) {
        return value;
    }
}

// 代理合约
contract Proxy {
    address public implementation;
    address public admin;

    constructor(address _implementation) {
        implementation = _implementation;
        admin = msg.sender;
    }

    // 升级逻辑合约
    function upgrade(address newImplementation) public {
        require(msg.sender == admin, "Not admin");
        implementation = newImplementation;
    }

    // Fallback：将所有调用转发到逻辑合约
    fallback() external payable {
        address impl = implementation;
        assembly {
            // 复制 calldata
            calldatacopy(0, 0, calldatasize())

            // delegatecall 到逻辑合约
            let result := delegatecall(gas(), impl, 0, calldatasize(), 0, 0)

            // 复制返回数据
            returndatacopy(0, 0, returndatasize())

            switch result
            case 0 { revert(0, returndatasize()) }
            default { return(0, returndatasize()) }
        }
    }

    receive() external payable {}
}
```

### 3. 拉取支付模式（Pull Payment）
```solidity
// ❌ 推送模式（危险）
contract PushPayment {
    address[] public recipients;

    function distributeRewards() public {
        for (uint256 i = 0; i < recipients.length; i++) {
            // 如果某个转账失败，整个函数 revert
            payable(recipients[i]).transfer(1 ether);  // 危险！
        }
    }
}

// ✅ 拉取模式（安全）
contract PullPayment {
    mapping(address => uint256) public pendingWithdrawals;

    function addReward(address recipient, uint256 amount) internal {
        pendingWithdrawals[recipient] += amount;
    }

    // 用户主动提取
    function withdraw() public {
        uint256 amount = pendingWithdrawals[msg.sender];
        require(amount > 0, "No pending withdrawal");

        pendingWithdrawals[msg.sender] = 0;  // 先清零（防重入）
        payable(msg.sender).transfer(amount);
    }

    function getWithdrawable() public view returns (uint256) {
        return pendingWithdrawals[msg.sender];
    }
}
```

### 4. 紧急停止模式（Circuit Breaker）
```solidity
contract CircuitBreaker {
    bool public stopped = false;
    address public owner;

    modifier stopInEmergency() {
        require(!stopped, "Contract is paused");
        _;
    }

    modifier onlyInEmergency() {
        require(stopped, "Not in emergency");
        _;
    }

    modifier onlyOwner() {
        require(msg.sender == owner, "Not owner");
        _;
    }

    constructor() {
        owner = msg.sender;
    }

    function toggleCircuitBreaker() public onlyOwner {
        stopped = !stopped;
    }

    // 正常操作
    function deposit() public payable stopInEmergency {
        // 只有未暂停时可执行
    }

    // 紧急操作
    function emergencyWithdraw() public onlyInEmergency {
        // 只有暂停时可执行
        payable(msg.sender).transfer(address(this).balance);
    }
}
```

### 5. 检查-效果-交互模式（Checks-Effects-Interactions）
```solidity
contract CEIPattern {
    mapping(address => uint256) public balances;

    // ❌ 错误：交互在前（重入风险）
    function withdrawBad(uint256 amount) public {
        payable(msg.sender).transfer(amount);  // 交互
        balances[msg.sender] -= amount;        // 效果（危险！）
    }

    // ✅ 正确：检查-效果-交互
    function withdrawGood(uint256 amount) public {
        // 1. 检查（Checks）
        require(balances[msg.sender] >= amount, "Insufficient balance");

        // 2. 效果（Effects）
        balances[msg.sender] -= amount;

        // 3. 交互（Interactions）
        payable(msg.sender).transfer(amount);
    }
}
```

### 6. 速率限制模式（Rate Limiting）
```solidity
contract RateLimit {
    uint256 public constant LIMIT = 1 ether;
    uint256 public constant PERIOD = 1 days;

    mapping(address => uint256) public lastWithdrawTime;
    mapping(address => uint256) public withdrawnInPeriod;

    function withdraw(uint256 amount) public {
        // 检查是否过了一个周期
        if (block.timestamp >= lastWithdrawTime[msg.sender] + PERIOD) {
            withdrawnInPeriod[msg.sender] = 0;
            lastWithdrawTime[msg.sender] = block.timestamp;
        }

        // 检查是否超过限额
        require(
            withdrawnInPeriod[msg.sender] + amount <= LIMIT,
            "Exceeds daily limit"
        );

        withdrawnInPeriod[msg.sender] += amount;
        payable(msg.sender).transfer(amount);
    }

    function getRemainingLimit(address user) public view returns (uint256) {
        if (block.timestamp >= lastWithdrawTime[user] + PERIOD) {
            return LIMIT;
        }
        return LIMIT - withdrawnInPeriod[user];
    }
}
```

## 9.5 实战：完整的 ERC20 代币

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

library SafeMath {
    function add(uint256 a, uint256 b) internal pure returns (uint256) {
        uint256 c = a + b;
        require(c >= a, "SafeMath: addition overflow");
        return c;
    }

    function sub(uint256 a, uint256 b) internal pure returns (uint256) {
        require(b <= a, "SafeMath: subtraction overflow");
        return a - b;
    }
}

contract MyERC20 {
    using SafeMath for uint256;

    string public name = "MyToken";
    string public symbol = "MTK";
    uint8 public decimals = 18;
    uint256 public totalSupply;

    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    event Transfer(address indexed from, address indexed to, uint256 value);
    event Approval(address indexed owner, address indexed spender, uint256 value);

    constructor(uint256 _initialSupply) {
        totalSupply = _initialSupply * 10**uint256(decimals);
        balanceOf[msg.sender] = totalSupply;
        emit Transfer(address(0), msg.sender, totalSupply);
    }

    function transfer(address _to, uint256 _value) public returns (bool) {
        require(_to != address(0), "Invalid address");
        require(balanceOf[msg.sender] >= _value, "Insufficient balance");

        balanceOf[msg.sender] = balanceOf[msg.sender].sub(_value);
        balanceOf[_to] = balanceOf[_to].add(_value);

        emit Transfer(msg.sender, _to, _value);
        return true;
    }

    function approve(address _spender, uint256 _value) public returns (bool) {
        allowance[msg.sender][_spender] = _value;
        emit Approval(msg.sender, _spender, _value);
        return true;
    }

    function transferFrom(address _from, address _to, uint256 _value)
        public
        returns (bool)
    {
        require(_to != address(0), "Invalid address");
        require(balanceOf[_from] >= _value, "Insufficient balance");
        require(allowance[_from][msg.sender] >= _value, "Insufficient allowance");

        balanceOf[_from] = balanceOf[_from].sub(_value);
        balanceOf[_to] = balanceOf[_to].add(_value);
        allowance[_from][msg.sender] = allowance[_from][msg.sender].sub(_value);

        emit Transfer(_from, _to, _value);
        return true;
    }
}
```
